var RESTAURANTS = [

// ───────────────────────── Вспомогательное ─────────────────────────


function ss_() { return SpreadsheetApp.getActiveSpreadsheet(); }


function sheet_(name) {

  var s = ss_().getSheetByName(name);

  if (!s) throw new Error('Нет листа «' + name + '». Запустите функцию setup.');

  return s;

}


function out_(obj) {

  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);

}


function restName_(id) {

  for (var i = 0; i < RESTAURANTS.length; i++) if (RESTAURANTS[i].id === id) return RESTAURANTS[i].name;

  return null;

}


function withLock_(fn) {

  var lock = LockService.getScriptLock();

  lock.waitLock(25000);

  try { return fn(); } finally { lock.releaseLock(); }

}


function normDate_(v) {

  if (Object.prototype.toString.call(v) === '[object Date]') {

    return Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd');

  }

  return String(v).slice(0, 10);

}


function isTrue_(v) { return v === true || String(v).toUpperCase() === 'TRUE'; }


// ───────────────────────── Точки входа ─────────────────────────


function doGet(e) {

  return out_({ok: true, service: 'luckee-track', hint: 'Это сервер. Откройте сайт, а не эту ссылку.'});

}


function doPost(e) {

  var req;

  try { req = JSON.parse(e.postData.contents); }

  catch (err) { return out_({ok: false, error: 'bad_request'}); }

  try { return out_(route_(req)); }

  catch (err) { return out_({ok: false, error: 'server', message: String(err && err.message || err)}); }

}


function route_(req) {

  var action = req.action;

  if (action === 'login') return login_(req);

  var auth = checkAuth_(req.auth);

  if (!auth) return {ok: false, error: 'auth'};

  switch (action) {

    case 'getState':   return getState_(auth, req);

    case 'entryOps':   return withLock_(function () { return entryOps_(auth, req); });

    case 'catalogOps': return withLock_(function () { return catalogOps_(auth, req); });

    case 'getCodes':   return getCodes_(auth);

    case 'setCodes':   return withLock_(function () { return setCodes_(auth, req); });

  }

  return {ok: false, error: 'unknown_action'};

}


// ───────────────────────── Коды и права ─────────────────────────


var codeIsNum_ = {};

function readCodes_() {

  var s = sheet_(SH_CODES), n = s.getLastRow(), map = {};

  codeIsNum_ = {};

  if (n >= 2) {

    s.getRange(2, 1, n - 1, 2).getValues().forEach(function (r) {

      var k = String(r[0]).trim();

      if (!k) return;

      if (typeof r[1] === 'number') codeIsNum_[k] = true; // Таблица превратила "0000" в число 0

      map[k] = String(r[1]).trim();

    });

  }

  return map;

}


// Совпадение кода. Если в ячейке число (потеряны нули впереди) — сравниваем как числа.

function codeMatches_(stored, key, given) {

  if (!stored) return false;

  if (stored === given) return true;

  return !!codeIsNum_[key] && /^\d+$/.test(given) && Number(given) === Number(stored);

}


// Возвращает {tov: bool, restaurant: id|null} или null, если код неверный

function checkAuth_(auth) {

  if (!auth || !auth.code) return null;

  var given = String(auth.code).trim();

  if (!given) return null;

  var codes = readCodes_();

  if (auth.role === 'tovaroved') {

    return codeMatches_(codes[TOV_NAME], TOV_NAME, given) ? {tov: true, restaurant: null} : null;

  }

  if (auth.role === 'chef') {

    var name = restName_(auth.restaurant);

    return (name && codeMatches_(codes[name], name, given)) ? {tov: false, restaurant: auth.restaurant} : null;

  }

  return null;

}


function login_(req) {

  var cache = CacheService.getScriptCache();

  var fails = Number(cache.get('fails') || 0);

  if (fails >= 30) return {ok: false, error: 'throttled'};

  var a = checkAuth_({role: req.role, restaurant: req.restaurant, code: req.code});

  if (!a) {

    cache.put('fails', String(fails + 1), 60);

    return {ok: false, error: 'auth'};

  }

  return {ok: true};

}


function getCodes_(auth) {

  if (!auth.tov) return {ok: false, error: 'forbidden'};

  var codes = readCodes_(), out = {tovaroved: codes[TOV_NAME] || ''};

  RESTAURANTS.forEach(function (r) { out[r.id] = codes[r.name] || ''; });

  return {ok: true, codes: out};

}


function setCodes_(auth, req) {

  if (!auth.tov) return {ok: false, error: 'forbidden'};

  var input = req.codes || {};

  var pairs = [[TOV_NAME, input.tovaroved]];

  RESTAURANTS.forEach(function (r) { pairs.push([r.name, input[r.id]]); });

  var s = sheet_(SH_CODES);

  var n = s.getLastRow();

  var vals = n >= 2 ? s.getRange(2, 1, n - 1, 2).getValues() : [];

  pairs.forEach(function (p) {

    var code = String(p[1] == null ? '' : p[1]).trim();

    if (!code) return;

    if (code.length > 30) throw new Error('Код слишком длинный');

    var found = false;

    for (var i = 0; i < vals.length; i++) {

      if (String(vals[i][0]).trim() === p[0]) { vals[i][1] = code; found = true; break; }

    }

    if (!found) vals.push([p[0], code]);

  });

  if (vals.length) {

    s.getRange(2, 2, vals.length, 1).setNumberFormat('@');

    s.getRange(2, 1, vals.length, 2).setValues(vals);

  }

  return {ok: true};

}


// ───────────────────────── Состояние (чтение) ─────────────────────────


function getState_(auth, req) {

  var rid = req.restaurant;

  if (!restName_(rid)) return {ok: false, error: 'bad_restaurant'};

  if (!auth.tov && auth.restaurant !== rid) return {ok: false, error: 'forbidden'};

  var props = PropertiesService.getScriptProperties();

  var ev = String(props.getProperty('ev_' + rid) || '0');

  var cv = String(props.getProperty('cv') || '0');

  var res = {ok: true, entriesVersion: ev, catalogVersion: cv, entries: null, catalog: null};

  if (String(req.entriesVersion) !== ev) res.entries = readEntries_(rid);

  if (String(req.catalogVersion) !== cv) res.catalog = readCatalog_();

  return res;

}


function readEntries_(rid) {

  var s = sheet_(SH_ENTRIES), last = s.getLastRow(), out = [];

  if (last < 2) return out;

  var start = Math.max(2, last - ENTRY_ROWS_WINDOW + 1);

  var vals = s.getRange(start, 1, last - start + 1, 11).getValues();

  var name = restName_(rid);

  for (var i = 0; i < vals.length; i++) {

    var r = vals[i];

    if (!r[0] || String(r[2]) !== name) continue;

    out.push({

      id: String(r[0]), date: normDate_(r[1]), category: String(r[3]), name: String(r[4]),

      unit: String(r[5]), qty: Number(r[6]), author: String(r[7] || ''),

      archived: isTrue_(r[8]), trashed: isTrue_(r[9]), ts: Date.parse(String(r[10])) || 0

    });

  }

  return out;

}


function readCatalog_() {

  var s = sheet_(SH_CATALOG), last = s.getLastRow();

  if (last < 2) return [];

  var vals = s.getRange(2, 1, last - 1, 5).getValues();

  var missing = false;

  for (var i = 0; i < vals.length; i++) if (vals[i][2] && !vals[i][4]) { missing = true; break; }

  if (missing) {

    withLock_(fillMissingCatalogIds_);

    vals = s.getRange(2, 1, last - 1, 5).getValues();

  }

  var out = [];

  vals.forEach(function (r) {

    if (!r[2] || !r[4]) return;

    out.push({id: String(r[4]), category: String(r[0]).trim(), article: String(r[1] || '').trim(),

              name: String(r[2]).trim(), unit: String(r[3]).trim()});

  });

  return out;

}


// Если вы вставили новые строки в лист «Товары» руками — им нужен ID. Присваиваем.

function fillMissingCatalogIds_() {

  var s = sheet_(SH_CATALOG), last = s.getLastRow();

  if (last < 2) return;

  var vals = s.getRange(2, 1, last - 1, 5).getValues();

  var used = {};

  vals.forEach(function (r) { if (r[4]) used[String(r[4])] = true; });

  var n = 0, changed = false;

  vals.forEach(function (r) {

    if (r[2] && !r[4]) {

      var id;

      do { id = 'n' + (Date.now().toString(36)) + (n++); } while (used[id]);

      used[id] = true; r[4] = id; changed = true;

    }

  });

  if (changed) {

    s.getRange(2, 5, vals.length, 1).setNumberFormat('@');

    s.getRange(2, 1, vals.length, 5).setValues(vals);

  }

}


// В Google Таблицах нельзя писать за пределы существующих строк листа — сначала расширяем.

function ensureRows_(s, needed) {

  var max = s.getMaxRows();

  if (max < needed) s.insertRowsAfter(max, needed - max);

}


function bump_(key) {

  var p = PropertiesService.getScriptProperties();

  p.setProperty(key, String(Number(p.getProperty(key) || 0) + 1));

}

// req: {restaurant, add:[{id,date,category,name,unit,qty,author}], update:[{id,fields:{archived,trashed}}], del:[id]}


function entryOps_(auth, req) {

  var rid = req.restaurant, rname = restName_(rid);

  if (!rname) return {ok: false, error: 'bad_restaurant'};

  if (!auth.tov && auth.restaurant !== rid) return {ok: false, error: 'forbidden'};

  var adds = req.add || [], updates = req.update || [], dels = req.del || [];

  if (!auth.tov && updates.length) return {ok: false, error: 'forbidden'};


  var s = sheet_(SH_ENTRIES);

  var last = s.getLastRow();

  var ids = last >= 2 ? s.getRange(2, 1, last - 1, 1).getValues() : [];

  var rowOf = {};

  for (var i = 0; i < ids.length; i++) rowOf[String(ids[i][0])] = i + 2;


  // 1) добавление

  var rows = [];

  adds.forEach(function (a) {

    var qty = Number(a.qty);

    var id = String(a.id || '');

    if (!(qty > 0) || qty > 1e6) throw new Error('Некорректное количество');

    if (CATEGORIES.indexOf(a.category) < 0) throw new Error('Некорректная категория');

    if (UNITS.indexOf(a.unit) < 0) throw new Error('Некорректная единица');

    if (!a.name || String(a.name).length > 200) throw new Error('Некорректное название');

    if (!/^\d{4}-\d{2}-\d{2}$/.test(String(a.date))) throw new Error('Некорректная дата');

    if (!/^[A-Za-z0-9_\-]{1,60}$/.test(id)) throw new Error('Некорректный id');

    if (rowOf[id]) return; // уже есть — повторную отправку игнорируем

    rows.push([id, String(a.date), rname, a.category, String(a.name), a.unit, qty,

               String(a.author || '').slice(0, 40), false, false, new Date().toISOString()]);

    rowOf[id] = -1;

  });

  if (rows.length) {

    ensureRows_(s, last + rows.length);

    s.getRange(last + 1, 1, rows.length, 11).setValues(rows);

    last += rows.length;

  }


  // 2) флаги «внесено» / «в корзине» (только товаровед)

  if (updates.length && last >= 2) {

    var rest = s.getRange(2, 3, last - 1, 1).getValues();

    var flags = s.getRange(2, 9, last - 1, 2).getValues();

    var touched = false;

    updates.forEach(function (u) {

      var row = rowOf[String(u.id)];

      if (!(row > 0) || String(rest[row - 2][0]) !== rname) return;

      var f = u.fields || {};

      if (f.archived !== undefined) { flags[row - 2][0] = !!f.archived; touched = true; }

      if (f.trashed !== undefined) { flags[row - 2][1] = !!f.trashed; touched = true; }

    });

    if (touched) s.getRange(2, 9, flags.length, 2).setValues(flags);

  }


  // 3) окончательное удаление

  if (dels.length && last >= 2) {

    var rest2 = s.getRange(2, 3, last - 1, 1).getValues();

    var fl2 = s.getRange(2, 9, last - 1, 1).getValues();

    var toDelete = [];

    dels.forEach(function (id) {

      var row = rowOf[String(id)];

      if (!(row > 0) || String(rest2[row - 2][0]) !== rname) return;

      if (!auth.tov && isTrue_(fl2[row - 2][0])) return; // шеф не удаляет уже внесённое

      toDelete.push(row);

    });

    toDelete.sort(function (a, b) { return b - a; });

    var k = 0;

    while (k < toDelete.length) { // склеиваем подряд идущие строки

      var end = toDelete[k], start = end;

      while (k + 1 < toDelete.length && toDelete[k + 1] === start - 1) { start = toDelete[++k]; }

      s.deleteRows(start, end - start + 1);

      k++;

    }

  }


  bump_('ev_' + rid);

  var p = PropertiesService.getScriptProperties();

  return {ok: true, entries: readEntries_(rid), entriesVersion: String(p.getProperty('ev_' + rid))};

}


// ───────────────────────── Каталог (запись, только товаровед) ─────────────────────────

// req: {add:[{id,category,name,unit,article}], update:[{id,fields:{name,unit,article}}], del:[id]}


function catalogOps_(auth, req) {

  if (!auth.tov) return {ok: false, error: 'forbidden'};

  var s = sheet_(SH_CATALOG);

  fillMissingCatalogIds_();

  var last = s.getLastRow();

  var vals = last >= 2 ? s.getRange(2, 1, last - 1, 5).getValues() : [];

  var rowOf = {}, dup = {};

  vals.forEach(function (r, i) {

    rowOf[String(r[4])] = i;

    dup[String(r[0]).trim() + '|' + String(r[2]).trim() + '|' + String(r[3]).trim()] = true;

  });


  var newRows = [];

  (req.add || []).forEach(function (a) {

    var id = String(a.id || '');

    if (CATEGORIES.indexOf(a.category) < 0) throw new Error('Некорректная категория');

    if (UNITS.indexOf(a.unit) < 0) throw new Error('Некорректная единица');

    var name = String(a.name || '').trim();

    if (!name || name.length > 200) throw new Error('Некорректное название');

    if (!/^[A-Za-z0-9_\-]{1,60}$/.test(id) || rowOf[id] !== undefined) return;

    var key = a.category + '|' + name + '|' + a.unit;

    if (dup[key]) return;

    dup[key] = true;

    newRows.push([a.category, String(a.article || '').trim(), name, a.unit, id]);

  });


  var changed = false;

  (req.update || []).forEach(function (u) {

    var i = rowOf[String(u.id)];

    if (i === undefined) return;

    var f = u.fields || {};

    if (f.name !== undefined && String(f.name).trim()) vals[i][2] = String(f.name).trim();

    if (f.unit !== undefined && UNITS.indexOf(f.unit) >= 0) vals[i][3] = f.unit;

    if (f.article !== undefined) vals[i][1] = String(f.article || '').trim();

    changed = true;

  });

  if (changed) {

    s.getRange(2, 2, vals.length, 1).setNumberFormat('@');

    s.getRange(2, 1, vals.length, 5).setValues(vals);

  }


  var toDelete = [];

  (req.del || []).forEach(function (id) { var i = rowOf[String(id)]; if (i !== undefined) toDelete.push(i + 2); });

  toDelete.sort(function (a, b) { return b - a; });

  var k = 0;

  while (k < toDelete.length) {

    var end = toDelete[k], start = end;

    while (k + 1 < toDelete.length && toDelete[k + 1] === start - 1) { start = toDelete[++k]; }

    s.deleteRows(start, end - start + 1);

    k++;

  }


  if (newRows.length) {

    var now = s.getLastRow();

    ensureRows_(s, now + newRows.length);

    s.getRange(now + 1, 2, newRows.length, 1).setNumberFormat('@');

    s.getRange(now + 1, 1, newRows.length, 5).setValues(newRows);

  }


  bump_('cv');

  var p = PropertiesService.getScriptProperties();

  return {ok: true, catalog: readCatalog_(), catalogVersion: String(p.getProperty('cv'))};

}


// ───────────────────────── Первичная настройка ─────────────────────────


function ensureSheet_(name, headers) {

  var ss = ss_(), s = ss.getSheetByName(name);

  if (s) {

    var cur = s.getLastRow() >= 1 ? s.getRange(1, 1, 1, headers.length).getValues()[0] : [];

    var same = headers.every(function (h, i) { return String(cur[i]).trim() === h; });

    if (same) return {sheet: s, created: false};

    // Лист с таким именем уже есть, но устроен иначе — не удаляем, а переименовываем

    s.setName(name + ' (старый ' + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd.MM HH-mm') + ')');

  }

  s = ss.insertSheet(name);

  s.getRange(1, 1, 1, headers.length).setValues([headers]);

  s.getRange(1, 1, 1, headers.length).setFontWeight('bold');

  s.setFrozenRows(1);

  return {sheet: s, created: true};

}


/** Запустите ОДИН раз. Безопасно запускать повторно: данные не затираются. */

function setup() {

  var log = [];


  // Товары

  var c = ensureSheet_(SH_CATALOG, CAT_HEADERS);

  c.sheet.getRange(1, 2, c.sheet.getMaxRows(), 1).setNumberFormat('@');

  if (c.sheet.getLastRow() < 2) {

    var rows = SEED_CATALOG.map(function (r, i) { return [r[0], r[1], r[2], r[3], 'p' + (i + 1)]; });

    ensureRows_(c.sheet, rows.length + 1);

    c.sheet.getRange(2, 1, rows.length, 5).setValues(rows);

    log.push('Каталог заполнен: ' + rows.length + ' позиций');

  } else {

    log.push('Каталог уже заполнен — не тронут');

  }


  // Записи

  var e = ensureSheet_(SH_ENTRIES, ENT_HEADERS);

  e.sheet.getRange(1, 2, e.sheet.getMaxRows(), 1).setNumberFormat('@');   // Дата

  e.sheet.getRange(1, 11, e.sheet.getMaxRows(), 1).setNumberFormat('@');  // Время

  log.push(e.created ? 'Лист «Записи» создан' : 'Лист «Записи» уже есть');


  // Коды

  var k = ensureSheet_(SH_CODES, CODE_HEADERS);

  k.sheet.getRange(1, 2, k.sheet.getMaxRows(), 1).setNumberFormat('@');

  var n = k.sheet.getLastRow();

  var vals = n >= 2 ? k.sheet.getRange(2, 1, n - 1, 2).getValues() : [];

  var have = {};

  vals.forEach(function (r) { have[String(r[0]).trim()] = true; });

  var added = 0;

  Object.keys(DEFAULT_CODE_BY_NAME).forEach(function (name) {

    if (!have[name]) { vals.push([name, DEFAULT_CODE_BY_NAME[name]]); added++; }

  });

  // Если коды раньше были записаны как числа (0000 → 0), возвращаем им текстовый вид

  vals.forEach(function (r) {

    if (typeof r[1] === 'number') {

      var def = DEFAULT_CODE_BY_NAME[String(r[0]).trim()];

      r[1] = (def !== undefined && Number(def) === r[1]) ? def : String(r[1]);

    }

  });

  if (vals.length) k.sheet.getRange(2, 1, vals.length, 2).setValues(vals);

  log.push('Коды: добавлено ' + added + ' (стартовые: товаровед 0000, рестораны 1111–4444 — СМЕНИТЕ их в приложении)');


  var msg = log.join('\n');

  Logger.log(msg);
